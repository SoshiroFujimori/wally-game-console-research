cmake_minimum_required(VERSION 3.14)
project(command_transport_study LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(WALLY_REPO "" CACHE PATH "Unmodified published console checkout")
if(NOT EXISTS "${WALLY_REPO}/examples/rasterix/CMakeLists.txt")
    message(FATAL_ERROR "Set WALLY_REPO to the published console checkout")
endif()
add_subdirectory(${WALLY_REPO}/examples/rasterix cvw-example)
add_executable(transport-study target.cpp)
target_include_directories(transport-study PRIVATE ${WALLY_REPO}/examples/rasterix)
target_link_libraries(transport-study PRIVATE gl span spdlog::spdlog threadrunner)
target_link_options(transport-study PRIVATE -static)
add_executable(transport-texture texture-probe.cpp)
target_include_directories(transport-texture PRIVATE ${WALLY_REPO}/examples/rasterix)
target_link_libraries(transport-texture PRIVATE gl span spdlog::spdlog threadrunner)
target_link_options(transport-texture PRIVATE -static)
option(TRANSPORT_BUILD_MODEL "Build a native full-RTL functional test" OFF)
if(TRANSPORT_BUILD_MODEL)
    find_package(verilator REQUIRED HINTS /usr/local/share/verilator)
    set(RIX_RTL ${WALLY_REPO}/addins/rasterix/rtl)
    set(TRANSPORT_MODEL_TOP "${RIX_RTL}/RasterIX/RasterIX_IF.v" CACHE FILEPATH "Official or explicitly experimental model top")
    get_filename_component(TRANSPORT_MODEL_INCLUDE ${TRANSPORT_MODEL_TOP} DIRECTORY)
    add_executable(gpu-functional gpu_functional.cpp)
    add_executable(gpu-texture texture-probe.cpp)
    target_compile_definitions(gpu-texture PRIVATE TRANSPORT_MODEL)
    foreach(model_target gpu-functional gpu-texture)
    target_link_libraries(${model_target} PRIVATE gl span spdlog::spdlog threadrunner)
    verilate(${model_target} TOP_MODULE RasterIX_IF PREFIX VRasterIX_IF
        SOURCES ${TRANSPORT_MODEL_TOP}
        INCLUDE_DIRS ${TRANSPORT_MODEL_INCLUDE} ${CMAKE_CURRENT_SOURCE_DIR}/vendor ${RIX_RTL}/RasterIX ${RIX_RTL}/Util ${RIX_RTL}/Float/rtl/float ${RIX_RTL}/3rdParty
            ${RIX_RTL}/3rdParty/verilog-axi ${RIX_RTL}/3rdParty/verilog-axis
        VERILATOR_ARGS -Wno-fatal -DUNITTEST -GTMU_COUNT=1 -GDATA_WIDTH=64 -GID_WIDTH=4
            -GFRAMEBUFFER_SIZE_IN_PIXEL_LG=${RIX_CORE_FRAMEBUFFER_SIZE_IN_PIXEL_LG})
    endforeach()
endif()
